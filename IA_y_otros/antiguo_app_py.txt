    "Ecuaciones No Lineales"
])

with tab_guia:
    st.header("Guía para ingresar ecuaciones")
    st.write("Para que la calculadora interprete correctamente las funciones matemáticas, utilice la siguiente sintaxis de programación:")
    
    st.markdown(r"""

    | **Trigonométricas Inversas** | Arco seno ($\arcsin x$) | `asin(x)` o `arcsin(x)` | `asin(x)` para $\arcsin(x)$ |
    | | Arco coseno ($\arccos x$) | `acos(x)` o `arccos(x)` | `acos(x)` para $\arccos(x)$ |

    st.info("**Nota importante para el Método de Punto Fijo:**\n\nEl método requiere que usted despeje la variable $x$ previamente de la ecuación original $f(x)=0$ para obtener la forma $x = g(x)$. **En la pestaña de cálculo, usted debe ingresar únicamente la función $g(x)$ resultante de ese despeje, no el $f(x)$ original.**")

# -----------------------------------------------------------------------------
# PESTAÑA 1: ANÁLISIS DE ERRORES
# -----------------------------------------------------------------------------
with tab_errores:
    col_opt1, col_opt2 = st.columns([2, 1])
    with col_opt1:
        opcion_error = st.selectbox(
            "Seleccione el tipo de cálculo de error:",
            [
                "Error absoluto",
                "Error relativo",
                "Error porcentual",
                "Error de redondeo",
                "Error de truncamiento",
                "Error en funciones",
                "Todos"
            ]
        )
    with col_opt2:
        num_dec = st.number_input("Decimales de precisión:", min_value=1, max_value=12, value=6, step=1)

    if opcion_error == "Error absoluto":
        col1, col2 = st.columns(2)
        with col1:
            val_x = st.text_input("Valor real (x)", value="", placeholder="Digite el valor")
        with col2:
            val_x_bar = st.text_input("Valor aproximado (x̄)", value="", placeholder="Digite el valor")

        latex_x = val_x.strip() if val_x.strip() != "" else r"x"
        latex_x_bar = val_x_bar.strip() if val_x_bar.strip() != "" else r"\bar{x}"

        st.latex(rf"|E_A| = |{latex_x} - {latex_x_bar}|")

        try:
            x_num = float(val_x)
            x_bar_num = float(val_x_bar)
            e_a = abs(x_num - x_bar_num)
            st.latex(rf"|E_A| = {e_a:.{num_dec}f}")
        except ValueError:
            pass

    elif opcion_error == "Error relativo":
        col1, col2 = st.columns(2)
        with col1:
            val_x = st.text_input("Valor real (x)", value="", placeholder="Digite el valor")
        with col2:
            val_x_bar = st.text_input("Valor aproximado (x̄)", value="", placeholder="Digite el valor")

        latex_x = val_x.strip() if val_x.strip() != "" else r"x"
        latex_x_bar = val_x_bar.strip() if val_x_bar.strip() != "" else r"\bar{x}"

        st.latex(rf"|E_R| = \frac{{|E_A|}}{{|x|}} = \frac{{|{latex_x}-{latex_x_bar}|}}{{|{latex_x}|}}")

        try:
            x_num = float(val_x)
            x_bar_num = float(val_x_bar)
            if x_num == 0:
                st.error("El valor real (x) no puede ser 0.")
            else:
                e_a = abs(x_num - x_bar_num)
                e_r = e_a / abs(x_num)
                st.latex(rf"|E_R| = \frac{{|{e_a:.{num_dec}f}|}}{{|{x_num:.{num_dec}f}|}}")
                st.latex(rf"|E_R| = {e_r:.{num_dec}f}")
        except ValueError:
            pass

    elif opcion_error == "Error porcentual":
        col1, col2 = st.columns(2)
        with col1:
            val_x = st.text_input("Valor real (x)", value="", placeholder="Digite el valor")
        with col2:
            val_x_bar = st.text_input("Valor aproximado (x̄)", value="", placeholder="Digite el valor")

        latex_x = val_x.strip() if val_x.strip() != "" else r"x"
        latex_x_bar = val_x_bar.strip() if val_x_bar.strip() != "" else r"\bar{x}"

        st.latex(rf"|E_P| = \frac{{|{latex_x} - {latex_x_bar}|}}{{|{latex_x}|}} \cdot 100")

        if val_x.strip() and val_x_bar.strip():
            try:
                x_num = float(val_x)
                x_bar_num = float(val_x_bar)

                if x_num == 0:
                    st.error("El valor real (x) no puede ser 0 para calcular el error relativo y porcentual.")
                else:
                    e_a = abs(x_num - x_bar_num)
                    st.latex(rf"|E_A| = |{x_num:.{num_dec}f} - {x_bar_num:.{num_dec}f}| = {e_a:.{num_dec}f}")

                    e_r = e_a / abs(x_num)
                    st.latex(rf"|E_R| = \frac{{|{e_a:.{num_dec}f}|}}{{|{x_num:.{num_dec}f}|}} = {e_r:.{num_dec}f}")

                    e_p = e_r * 100
                    st.latex(rf"|E_P| = {e_r:.{num_dec}f} \cdot 100 = {e_p:.{num_dec}f}\%")

            except ValueError:
                st.error("Por favor, ingrese valores numéricos válidos (use punto '.' para decimales).")

    elif opcion_error == "Error de redondeo":
        col1, col2, col3 = st.columns(3)
        with col1:
            val_x = st.text_input("Valor real (x)", value="", placeholder="Digite el valor")
        with col2:
            val_x_bar = st.text_input("Valor aproximado (x̄)", value="", placeholder="Digite el valor")
        with col3:
            val_d = st.text_input("Cifras significativas (d)", value="", placeholder="Ej: 3")

        latex_x = val_x.strip() if val_x.strip() != "" else r"x"
        latex_x_bar = val_x_bar.strip() if val_x_bar.strip() != "" else r"\bar{x}"
        latex_d = val_d.strip() if val_d.strip() != "" else r"d"

        st.latex(rf"|E_R^d| = |{latex_x} - {latex_x_bar}| \le \frac{{1}}{{2}} \cdot 10^{{-{latex_d}}}")

        try:
            if val_d.strip() != "":
                d_num = int(val_d)
                cota_redondeo = 0.5 * (10 ** -d_num)
                st.latex(rf"\text{{Cota máxima de error para }} d = {d_num}: \quad \frac{{1}}{{2}} \cdot 10^{{-{d_num}}} = {cota_redondeo:.{num_dec}f}")

                if val_x.strip() != "" and val_x_bar.strip() != "":
                    x_num = float(val_x)
                    x_bar_num = float(val_x_bar)
                    e_a = abs(x_num - x_bar_num)

                    st.latex(rf"|E_R^{{{d_num}}}| = |{x_num:.{num_dec}f} - {x_bar_num:.{num_dec}f}| = {e_a:.{num_dec}f} \le {cota_redondeo:.{num_dec}f}")

                    if e_a <= cota_redondeo:
                        st.success(f"¡Cumple! El error absoluto ({e_a:.{num_dec}f}) es menor o igual a la cota teórica ({cota_redondeo:.{num_dec}f}).")
                        st.info(f"**Esto significa que tu redondeo de {latex_d} decimales es correcto.**")
                    else:
                        st.error(f"No cumple. El error absoluto ({e_a:.{num_dec}f}) supera la cota teórica ({cota_redondeo:.{num_dec}f}) para {d_num} cifras.")

        except ValueError:
            st.error("Por favor, ingrese un número entero válido para las cifras significativas (d) y números válidos para x y x̄.")
            pass

    elif opcion_error == "Error de truncamiento":
        st.subheader("Error de Truncamiento mediante Serie de Taylor")

        col1, col2 = st.columns(2)
        with col1:
            func_str = st.text_input("Ingrese la función f(x):", value="exp(x)", placeholder="Ej: exp(x), sin(x)")
            val_x0 = st.text_input("Punto de expansión (x₀):", value="0", placeholder="Ej: 0")
        with col2:
            val_x = st.text_input("Valor a evaluar (x):", value="0.5", placeholder="Ej: 0.5")
            val_d = st.text_input("Orden / Términos de Taylor (d):", value="3", placeholder="Ej: 3")

        latex_d = val_d.strip() if val_d.strip() != "" else r"d"
        st.latex(rf"|E_T^{{{latex_d}}}| = |f(x) - P_{{{latex_d}}}(x)| \le 1 \cdot 10^{{-{latex_d}}}")

        if func_str.strip() and val_x.strip() and val_d.strip():
            try:
                x_sym = sp.Symbol('x')
                expr = sp.sympify(func_str)
                x_num = float(val_x)
                x0_num = float(val_x0) if val_x0.strip() != "" else 0.0
                d_num = int(val_d)

                # 1. Cálculo del valor real f(x)
                val_real = float(expr.subs(x_sym, x_num))

                # 2. Construcción de la Serie de Taylor e iteración hasta orden d
                taylor_poly = expr.series(x_sym, x0_num, n=d_num + 1).removeO()
                val_aprox = float(taylor_poly.subs(x_sym, x_num))

                # 3. Error de truncamiento y cota teórica
                e_t = abs(val_real - val_aprox)
                cota_truncamiento = 1.0 * (10 ** -d_num)

                st.markdown("**1. Polinomio de Taylor resultante $P_d(x)$:**")
                st.latex(rf"P_{{{d_num}}}(x) = {sp.latex(taylor_poly)}")

                st.markdown("**2. Evaluación:**")
                st.latex(rf"\text{{Valor Real }} f({x_num}) = {val_real:.{num_dec}f}")
                st.latex(rf"\text{{Valor Aproximado (Taylor)}} P_{{{d_num}}}({x_num}) = {val_aprox:.{num_dec}f}")
                st.latex(rf"|E_T^{{{d_num}}}| = |{val_real:.{num_dec}f} - {val_aprox:.{num_dec}f}| = {e_t:.{num_dec}f} \le {cota_truncamiento:.{num_dec}f}")

                # 4. Verificación alineada con la precisión visual seleccionada
                if round(e_t, num_dec) <= round(cota_truncamiento, num_dec):
                    st.success(f"¡Cumple! El error de truncamiento ({e_t:.{num_dec}f}) es menor o igual a la cota ({cota_truncamiento:.{num_dec}f}). La cantidad de decimales/términos 'd = {d_num}' elegida es correcta.")
                else:
                    st.error(f"No cumple. El error de truncamiento ({e_t:.{num_dec}f}) supera la cota teórica ({cota_truncamiento:.{num_dec}f}) para d = {d_num}. Se requieren más términos en la serie.")

            except Exception as e:
                st.error(f"Error al calcular la Serie de Taylor: {e}")
    elif opcion_error == "Error en funciones":
        st.subheader("Error en Funciones de Varias Variables (Propagación de Error)")
        
        # Entrada dinámica de la función
        func_expr_str = st.text_input(
            "Ingrese la función f(x1, x2, ...):", 
            value="3*x1*x2**3", 
            placeholder="Ej: 3*x1*x2**3  o  2**x * y + exp(y) + log(y)"
        )

        if func_expr_str.strip():
            try:
                # Interpretación matemática simbólica
                expr = sp.sympify(func_expr_str)
                variables = sorted(list(expr.free_symbols), key=lambda s: s.name)

                if not variables:
                    st.warning("Ingrese una función que contenga al menos una variable.")
                else:
                    st.markdown("**Símbolos detectados en la función:** " + ", ".join([f"`{v.name}`" for v in variables]))
                    
                    # Formulario dinámico para capturar el valor real y el error de cada variable
                    valores_nominales = {}
                    errores_variables = {}

                    st.markdown("---")
                    st.markdown("### Valores de entrada y márgenes de error")
                    
                    cols = st.columns(len(variables))
                    for idx, var in enumerate(variables):
                        with cols[idx]:
                            v_val = st.text_input(f"Valor ({var.name}):", value="2.0" if idx==0 else "3.0")
                            v_err = st.text_input(f"Error EA({var.name}):", value="0.01")
                            
                            if v_val.strip() and v_err.strip():
                                valores_nominales[var] = float(v_val)
                                errores_variables[var] = float(v_err)

                    # Procesar solo cuando todas las variables tengan sus datos cargados
                    if len(valores_nominales) == len(variables):
                        st.markdown("---")
                        
                        # PASO 1: Valoración de la función original
                        val_pares = [(var, valores_nominales[var]) for var in variables]
                        y_0 = float(expr.subs(val_pares))
                        
                        st.markdown("**Paso 1: Valoración (remplazando en la función original):**")
                        st.latex(rf"f = {sp.latex(expr)} = {y_0:.{num_dec}f}")

                        # PASO 2: Derivadas Parciales y PASO 3: Propagación de Error
                        st.markdown("**Paso 2 y 3: Derivadas Parciales y Evaluación del Error Total:**")
                        
                        e_a_total = 0.0
                        terminos_latex = []

                        for var in variables:
                            # Derivada parcial respecto a la variable actual
                            df_var = sp.diff(expr, var)
                            df_val = float(df_var.subs(val_pares))
                            err_var = errores_variables[var]
                            
                            # Acumulativo del error total: |df/dvar| * EA(var)
                            aporte_error = abs(df_val) * err_var
                            e_a_total += aporte_error

                            st.latex(rf"\frac{{\partial f}}{{\partial {var.name}}} = {sp.latex(df_var)} \quad \Rightarrow \quad \left|{df_val:.{num_dec}f}\right| \cdot {err_var} = {aporte_error:.{num_dec}f}")
                            terminos_latex.append(rf"\left|\frac{{\partial f}}{{\partial {var.name}}}\right| \cdot E_{{A({var.name})}}")

                        # Fórmula de suma acumulada en LaTeX
                        formula_suma = " + ".join(terminos_latex)
                        st.latex(rf"E_{{A(total)}} = {formula_suma} = {e_a_total:.{num_dec}f}")

                        # PASO 4: Expresión del intervalo final
                        st.markdown("---")
                        st.markdown("**Paso 4: Resultado final (Valor ± Error Total):**")
                        st.latex(rf"Resultado = {y_0:.{num_dec}f} \pm {e_a_total:.{num_dec}f}")

            except Exception as e:
                st.error(f"Error al evaluar la expresión matemática: {e}")
    elif opcion_error == "Todos":
        col1, col2, col3 = st.columns(3)
        with col1:
            val_real = st.number_input("Valor Real (x)", value=1.206539823, format="%.9f")
        with col2:
            val_aprox = st.number_input("Valor Aproximado (x*)", value=1.206433, format="%.6f")
        with col3:
            d_cifras = st.number_input("Cifras significativas (d)", value=3, min_value=1, step=1)

        err_abs = abs(val_real - val_aprox)
        err_rel = (err_abs / abs(val_real)) if val_real != 0 else 0
        err_porcentual = err_rel * 100

        cota_redondeo = 0.5 * (10 ** -d_cifras)
        cumple_redondeo = err_abs <= cota_redondeo

        col_res1, col_res2 = st.columns(2)
        with col_res1:
            st.markdown(f"**Error Absoluto ($|E_A|$):** `{err_abs:.{num_dec}f}`")
            st.markdown(f"**Error Relativo ($|E_R|$):** `{err_rel:.{num_dec}f}`")
            st.markdown(f"**Error Porcentual ($|E_P|$):** `{err_porcentual:.{num_dec}f}%`")

        with col_res2:
            st.subheader("Error de redondeo")
            st.latex(r"|E_A^d| \le \frac{1}{2} \cdot 10^{-d}")
            st.write(f"Cota máxima para d={d_cifras}: **{cota_redondeo:.{num_dec}f}**")

        if cumple_redondeo:
            st.success(f"¡Cumple! El error absoluto ({err_abs:.{num_dec}f}) es menor o igual a la cota teórica ({cota_redondeo:.{num_dec}f}). Elegiste bien 'd'.")
        else:
            st.error(f"No cumple. El error absoluto ({err_abs:.{num_dec}f}) supera la cota teórica ({cota_redondeo:.{num_dec}f}) para {d_cifras} cifras.")

# -----------------------------------------------------------------------------
# PESTAÑA 2: ECUACIONES NO LINEALES
# -----------------------------------------------------------------------------
with tab_ecuaciones:
    metodo = st.selectbox("Seleccione el Método Numérico:", [
        "Método de Punto Fijo", 
        "Newton-Raphson (1er Orden)", 
        "Newton-Raphson (2do Orden)"
    ])

    col_input, col_geo = st.columns([1, 1])

    with col_input:
        if metodo == "Método de Punto Fijo":
            st.latex(r"x_{i+1} = g(x_i)")
            st.caption("Recuerda ingresar la función **g(x)** previamente despejada.")
            func_expr = st.text_input("Ingrese g(x):", value="(x**2 - exp(x)) / 5")
        elif metodo == "Newton-Raphson (1er Orden)":
            st.latex(r"x_{i+1} = x_i - \frac{f(x_i)}{f'(x_i)}")
            func_expr = st.text_input("Ingrese f(x):", value="x**2 - 5*x - exp(x)")
        else:
            st.latex(r"x_{i+1} = x_i - \frac{f(x_i) \cdot f'(x_i)}{[f'(x_i)]^2 - f(x_i) \cdot f''(x_i)}")
            func_expr = st.text_input("Ingrese f(x):", value="x**2 - 5*x - exp(x)")

        x0 = st.number_input("Valor inicial (x0):", value=0.0, format="%.4f")
        tol = st.number_input("Tolerancia del error (E):", value=0.001, format="%.5f")
        max_iter = st.slider("Máximo de iteraciones:", 1, 30, 10)

    with col_geo:
        st.subheader("GeoGebra")
        st.iframe("https://www.geogebra.org/classic", height=310)

    if st.button("Ejecutar Método"):
        x_sym = sp.Symbol("x")
        
        try:
            expr = sp.sympify(func_expr)
            
            st.markdown("### Función evaluada:")
            if metodo == "Método de Punto Fijo":
                st.latex(f"g(x) = {sp.latex(expr)}")
            else:
                st.latex(f"f(x) = {sp.latex(expr)}")

            if metodo == "Método de Punto Fijo":
                g_func = sp.lambdify(x_sym, expr, "numpy")
            else:
                f_func = sp.lambdify(x_sym, expr, "numpy")
                expr_d1 = sp.diff(expr, x_sym)
                f_d1_func = sp.lambdify(x_sym, expr_d1, "numpy")
                st.markdown("**Primera Derivada calculada automáticamente:**")
                st.latex(f"f'(x) = {sp.latex(expr_d1)}")
                
                if metodo == "Newton-Raphson (2do Orden)":
                    expr_d2 = sp.diff(expr_d1, x_sym)
                    f_d2_func = sp.lambdify(x_sym, expr_d2, "numpy")
                    st.markdown("**Segunda Derivada calculada automáticamente:**")
                    st.latex(f"f''(x) = {sp.latex(expr_d2)}")

            iteraciones = []
            x_curr = x0
            convergio = False

            for i in range(max_iter + 1):
                if metodo == "Método de Punto Fijo":
                    x_next = float(g_func(x_curr))
                
                elif metodo == "Newton-Raphson (1er Orden)":
                    f_val = float(f_func(x_curr))
                    fd1_val = float(f_d1_func(x_curr))
                    if fd1_val == 0:
                        st.error("Error matemático: La derivada es cero. División por cero evitada.")
                        break
                    x_next = x_curr - (f_val / fd1_val)
                
                elif metodo == "Newton-Raphson (2do Orden)":
                    f_val = float(f_func(x_curr))
                    fd1_val = float(f_d1_func(x_curr))
                    fd2_val = float(f_d2_func(x_curr))
                    
                    denominador = (fd1_val**2) - (f_val * fd2_val)
                    if denominador == 0:
                        st.error("Error matemático: El denominador se hizo cero en Newton de 2do Orden.")
                        break
                    
                    x_next = x_curr - ((f_val * fd1_val) / denominador)

                error = abs(x_next - x_curr)
                
                iteraciones.append({
                    "Iteración": i,
                    "x_i": x_curr,
                    "x_i+1": x_next,
                    "Error |x_i - x_i+1|": error
                })

                if error <= tol:
                    convergio = True
                    break
                    
                x_curr = x_next

            st.subheader("Tabla de Iteraciones")
            st.dataframe(iteraciones, width='stretch')

            if convergio:
                st.success(f"¡El método convergió exitosamente! La raíz aproximada es **{x_curr:.6f}**")
            else:
                st.warning("Se alcanzó el límite de iteraciones sin lograr la tolerancia fijada.")

        except Exception as e:
            st.error(f"Error en la evaluación. Por favor, verifique la sintaxis en la Guía de Símbolos. Detalle técnico: {e}")
